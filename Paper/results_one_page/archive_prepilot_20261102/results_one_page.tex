\section*{1\quad Axisymmetric reduced-MHD balances}
Use JOREK notation $p=\rho T$, $\partial_\phi=0$ and the case setting
$\tau_{IC}=0$. The field ansatz is
\begin{equation*}
\begin{aligned}
 \mathbf B&=\nabla\psi\times\nabla\phi+F_0\nabla\phi,\\
 \mathbf v&=-R^2\nabla u\times\nabla\phi+V_\parallel\mathbf B.
\end{aligned}
\end{equation*}
Selected balances and constraints are
\begin{equation}
\begin{aligned}
 \partial_t\rho &= \underbrace{-\nabla\!\cdot(\rho\mathbf v)}_{\mathcal A_\rho(q)}+\mathcal R_\rho(q),\\
 \partial_t p &= \underbrace{-\mathbf v\!\cdot\nabla p-\gamma p\nabla\!\cdot\mathbf v}_{\mathcal A_p(q)}+\mathcal R_p(q),\\
 \partial_t\psi &= R[\psi,u]+\mathcal R_\psi(q),\\
 z_j&=\Delta^*\psi,\qquad w=\Delta_{\rm pol}u.
\end{aligned}
\label{eq:mhd}
\end{equation}
Here $q=(\psi,u,z_j,w,\rho,T,V_\parallel)$ and
$[a,b]=a_Rb_Z-a_Zb_R$. The remainders retain transport, sources,
heating and resistivity/current drive. Momentum, Lorentz/pressure forces
and native temporal/weak operators stay numerical.

\section*{2\quad Physical terms selected for learning}
For $n=0$ and adiabatic index $\gamma$, the selected blocks are
\begin{equation}
\begin{aligned}
 \mathcal A_\rho(q)&=\frac{[R^2\rho,u]-[\rho V_\parallel,\psi]}{R},\\
 \mathcal A_p(q)&=R[p,u]+2\gamma p\,u_Z\\
 &\quad-\frac{V_\parallel[p,\psi]+\gamma p[V_\parallel,\psi]}{R}.
\end{aligned}
\label{eq:targets}
\end{equation}
\section*{3\quad Proposed neural-operator replacement}
Two independent L-SplineFNOs replace
$\mathcal A_\rho\mapsto\mathcal N_{\theta_\rho}$ and
$\mathcal A_p\mapsto\mathcal N_{\theta_p}$:
\begin{equation}
\begin{aligned}
 \partial_t\rho&=\mathcal N_{\theta_\rho}(q)+\mathcal R_\rho(q),\\
 \partial_t(\rho T)&=\mathcal N_{\theta_p}(q)+\mathcal R_p(q).
\end{aligned}
\label{eq:neural}
\end{equation}
Networks consume current fields at each integrator stage and return scalar
RHS terms at native quadrature queries.

With native weak assembly $\mathcal W$ and row injections $E_\rho,E_p$,
\begin{equation}
\begin{aligned}
 M(q)\dot q &= F_{\rm kept}(q;h)
       +E_\rho\mathcal W[\mathcal N_{\theta_\rho}(q)]\\
 &\quad+E_p\mathcal W[\mathcal N_{\theta_p}(q)],\qquad C(q)=0.
\end{aligned}
\label{eq:assembly}
\end{equation}
$M$ stays coupled: \textbf{a pressure residual is not $\dot T$}.
Magnetic/momentum dynamics, transport, boundaries, constraints and
step-dependent stabilization ($h$) remain numerical.
